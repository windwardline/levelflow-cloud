#!/usr/bin/env python3
"""Attribute every row of the 2026-09-24 re-simulate against the 2026-09-14 corpus.

Engine changes between them (expected-differences.md, written before the run):
  #641 2026.09.14.forex-commission-cross-rate: the 21 forex crosses' commission and everything
       downstream; a new usdPerQuote column on every row.
  #689 2026.09.23.stop-exit-slippage: stop-kind exits that did not gap print level - slippage.
  #692 2026.09.23.expiry-exit-slippage: the review-end close prints with slippage.

Every differing row is checked on two counts, cross or not:
  1. Its differing fields lie inside a named set: exit slippage and what follows from it on every
     row, and on a cross also the commission and what follows from that (cost, cost-derived scores,
     the reward ratios, acceptance). Any other field is a VIOLATION.
  2. If its legs differ: only exit legs moved, each by exactly estimatedSlippage against the position,
     and the row exits at a stop kind or the review-end close. Anything else is a VIOLATION.
On a cross the commission's own value is not recomputed; the check there is which fields moved, not by
how much. estimatedSlippage itself may not move anywhere.

Streams both files in lockstep by (symbol, time, variant, holdout), resyncing past rows present in only
one file. Confirm rows are withheld on their split before any other field is read. usdPerQuote, new with
#641, is stripped from each new row before the comparison. The forex roster is read from OLD's manifest,
beside it. Read-only; stdout only.
Usage: python3 attribute.py.txt OLD.jsonl NEW.jsonl [--limit N]
"""
import json, sys, collections

OLD, NEW = sys.argv[1], sys.argv[2]
LIMIT = int(sys.argv[sys.argv.index("--limit") + 1]) if "--limit" in sys.argv else None
MAJORS = {"EURUSD", "GBPUSD", "AUDUSD", "NZDUSD", "USDJPY", "USDCHF", "USDCAD"}
NEW_COLUMNS = {"usdPerQuote"}
STOP_KINDS = {"stop_loss", "breakeven_stop", "tp1_lock", "ambiguous", "same_bar_lock"}
# What exit slippage may move on any row: the exit leg prices, the R columns built on them, the
# labels that follow R's sign, and the arming-bound exit.
SLIPPAGE_FIELDS = {"legs", "realizedR", "armingBoundExitPrice", "armingBoundRealizedR",
                   "forgoneRunnerR", "outcome", "armingBoundOutcome"}
# What #641's commission may move on a cross, in addition.
COMMISSION_FIELDS = {"estimatedCommission", "estimatedRoundTripCost", "costShare", "rewardRisk",
                     "ladderRewardRisk", "grossRealizedR", "grossOutcome", "confidenceScore",
                     "executionScore", "accepted"}

def key(r):
    return (r["symbol"], r["time"], r["variant"], r.get("holdout"))

def exit_kinds(r):
    legs = r.get("legs") or []
    return {l.get("kind") for l in legs if l.get("leg") == "exit"}

counts = collections.Counter()
fields_by_group = {"cross": collections.Counter(), "noncross": collections.Counter()}
kinds_on_moved = {"cross": collections.Counter(), "noncross": collections.Counter()}
leg_moves = {"cross": collections.Counter(), "noncross": collections.Counter()}
unexpected_fields = collections.Counter()
violations = []
flips = collections.Counter()
only_old = collections.Counter(); only_new = collections.Counter()
with open(OLD + ".manifest.json", "r") as f:
    forex_symbols = {e["symbol"] for e in json.load(f)["symbols"] if e["assetType"] == "forex"}
assert len(forex_symbols) == 28, len(forex_symbols)

def legs_ok(o, n, group):
    ol, nl = o.get("legs") or [], n.get("legs") or []
    if len(ol) != len(nl):
        return False
    ok = True
    for a, b in zip(ol, nl):
        if a == b:
            continue
        if a.get("leg") != "exit" or {k: v for k, v in a.items() if k != "price"} != {k: v for k, v in b.items() if k != "price"}:
            ok = False
            leg_moves[group]["BAD:not an exit price"] += 1
            continue
        # The move must BE the modelled slippage (to the corpus's 8-decimal rounding), against the
        # position: a long exits lower, a short higher.
        d = b["price"] - a["price"]
        slip = o.get("estimatedSlippage") or 0
        against = (d < 0) if o["side"] == "buy" else (d > 0)
        if slip > 0 and abs(abs(d) - slip) <= 2e-8 + 1e-9 * abs(a["price"]) and against:
            leg_moves[group][a.get("kind")] += 1
        else:
            ok = False
            leg_moves[group][f"BAD:{a.get('kind')}"] += 1
    ek = exit_kinds(o) | exit_kinds(n)
    kinds_on_moved[group].update(ek or {"<no exit leg>"})
    if not (ek & (STOP_KINDS | {"expiry", "review_end"})) and not any("expir" in (k or "") for k in ek):
        ok = False
    return ok

def compare(o, n):
    # The door: confirm rows are withheld on their split before any other field is read.
    if o.get("split") == "confirm" or n.get("split") == "confirm":
        counts["confirm_withheld"] += 1
        return
    for c in NEW_COLUMNS:
        n.pop(c, None)
    if o == n:
        counts["identical"] += 1
        return
    cross = o["symbol"] in forex_symbols and o["symbol"] not in MAJORS
    group = "cross" if cross else "noncross"
    counts[f"differ_{group}"] += 1
    diff = sorted(k for k in set(o) | set(n) if o.get(k) != n.get(k))
    fields_by_group[group].update(diff)
    if o.get("accepted") != n.get("accepted"):
        flips[f"{group} {o.get('accepted')}->{n.get('accepted')}"] += 1
    allowed = SLIPPAGE_FIELDS | (COMMISSION_FIELDS if cross else set())
    stray = [k for k in diff if k not in allowed]
    ok = not stray
    unexpected_fields.update(stray)
    if "legs" in diff and not legs_ok(o, n, group):
        ok = False
    if not ok:
        counts[f"violation_{group}"] += 1
        if len(violations) < 25:
            violations.append({"key": key(o), "group": group, "diff": diff, "stray": stray,
                               "exitKinds": sorted(x or "" for x in exit_kinds(o) | exit_kinds(n))})

def rows(path):
    with open(path, "r") as f:
        for i, line in enumerate(f):
            if LIMIT is not None and i >= LIMIT:
                return
            yield json.loads(line)

oi, ni = rows(OLD), rows(NEW)
o = next(oi, None); n = next(ni, None)
pending_old, pending_new = {}, {}
while o is not None or n is not None:
    if o is not None and n is not None and key(o) == key(n):
        compare(o, n); o = next(oi, None); n = next(ni, None); continue
    # resync: park the unmatched row and look for its partner later
    if o is not None:
        ko = key(o)
        if ko in pending_new:
            compare(o, pending_new.pop(ko)); o = next(oi, None); continue
        pending_old[ko] = o; o = next(oi, None)
    if n is not None:
        kn = key(n)
        if kn in pending_old:
            compare(pending_old.pop(kn), n); n = next(ni, None); continue
        pending_new[kn] = n; n = next(ni, None)
    if len(pending_old) + len(pending_new) > 200_000:
        sys.exit("resync buffer exceeded 200k rows: the files are not in one order")
for k, r in pending_old.items(): only_old[r["symbol"]] += 1
for k, r in pending_new.items(): only_new[r["symbol"]] += 1

compared = counts["identical"] + counts["differ_cross"] + counts["differ_noncross"]
print(f"rows compared (fit and select): {compared}  confirm withheld at the door: {counts['confirm_withheld']}  {dict(counts)}")
print(f"only in old: {sum(only_old.values())} {dict(only_old.most_common(10))}")
print(f"only in new: {sum(only_new.values())} {dict(only_new.most_common(10))}")
print(f"accepted flips: {dict(flips)}")
for group in ("noncross", "cross"):
    print(f"{group} differing fields: {dict(fields_by_group[group].most_common())}")
    print(f"{group} exit kinds on rows whose legs moved: {dict(kinds_on_moved[group])}")
    print(f"{group} exit-leg price moves by kind: {dict(leg_moves[group])}")
print(f"fields outside the named sets: {dict(unexpected_fields)}")
print(f"VIOLATIONS: non-cross {counts['violation_noncross']}, cross {counts['violation_cross']}")
for v in violations: print("  ", v)
