/**
 * Mirror control, phase 2 — the same-geometry mirror of forex's shipped ladder.
 *
 * For each forex fit/select baseline accepted row (extracted by extract.mts
 * through the repo door), the shipped setup S is rebuilt from the row's own
 * fields and resolved with the engine's own `evaluateSetupOutcome` and the
 * engine's own leg accountant `realizedRFromLegs`, over the pinned 5-/15-minute
 * series at the manifest's anchor, sliced exactly as `simulateSymbol` slices
 * them (sweep.ts: resolutionSeriesFor; the 5-minute stream from the decision
 * bar's close, latest.time + 15 min, to latest.time + (defaultReviewHours + 24)
 * h; the 15-minute stream from the bar after the decision bar), with the cost
 * options from `resolverCostOptions` and `streamStartsAtMs` = time + 15 min.
 *
 * THE ANCHOR comes first and gates everything: S at the manifest's
 * modeledCostScale with same-bar arming ON must reproduce the row's outcome and
 * realizedR, arming OFF its armingBoundOutcome and armingBoundRealizedR, and at
 * the gross scale its grossRealizedR and grossOutcome — on every kept row,
 * unfilled included, to 1e-9. Any mismatch: the census is written, the mirror
 * figures are NOT, and the run exits 4.
 *
 * THE MIRROR M reflects every price level of S about the decision bar's close
 * C = row.latestClose (x -> 2C - x: entry, stop, TP1, target), takes the
 * opposite side, and keeps created_at, the review window, the runner
 * protection and the row's cost triple. Under a symmetric random walk S and M
 * are identically distributed.
 *
 * ARMS: FRICTIONLESS (resolverCostOptions over a zero triple: no half-spread,
 * no stop/gap/expiry slippage, and perLegCost 0 in the accountant) and FULL
 * COST (the row's triple at the manifest's modeledCostScale, commission/2 per
 * leg — exactly the corpus's realizedR / armingBoundRealizedR for S), each
 * with same-bar protection arming ON and OFF. Unfilled = 0 R per decision.
 *
 * ZERO PROVIDER BYTES: every series comes through q4's `pinnedSeries`, whose
 * fetchers throw; `fetch` itself is replaced by a thrower; the run refuses a
 * missing cache directory rather than creating one. Nothing is written under
 * the repository.
 *
 *   npx tsx <scratch>/mirror/resolve.mts <corpus.jsonl> <extractDir> <outDir> <cacheDir>
 */
(globalThis as { fetch: unknown }).fetch = () => {
  throw new Error("mirror: fetch is disabled — this reader spends zero provider bytes");
};

type Side = "buy" | "sell";
type Setup = { entry: number; side: Side; stop: number; target: number; tp1: number | null };
type Row = {
  split: "fit" | "select";
  time: number;
  side: Side;

  legs: ResolutionLeg[];
};
type Arm = { R: number; filled: boolean; outcome: string; legs: ResolutionLeg[]; feedback: Record<string, unknown>; exitAt: string; filledAt?: string };

const FIFTEEN = 15 * 60_000;
const FIVE = 5 * 60_000;
const HOUR = 3_600_000;
const DAY = 86_400_000;
const ZERO_TRIPLE = { estimatedCommission: 0, estimatedSlippage: 0, estimatedSpread: 0 };

function lowerBound(bars: readonly ReplayBar[], time: number): number {
  let low = 0;
  let high = bars.length;
  while (low < high) {
    const mid = (low + high) >> 1;
    if (bars[mid].time < time) low = mid + 1;
    else high = mid;
  }
  return low;
}
function upperBound(bars: readonly ReplayBar[], time: number): number {
  let low = 0;
  let high = bars.length;
  while (low < high) {
    const mid = (low + high) >> 1;
    if (bars[mid].time <= time) low = mid + 1;
    else high = mid;
  }
  return low;
}

function resolveArm(input: {
  symbol: string;
  row: Row;
  setup: Setup;
  bars: ReplayBar[];
  intervalMs: number;
  now: number;
  charged: boolean; // false = frictionless
  scale: number;
  arming: boolean;
}): Arm {
  const { symbol, row, setup, bars, intervalMs, now, charged, scale, arming } = input;
  const cal = calOf.get(symbol)!;
  const triple = charged
    ? { estimatedCommission: row.estimatedCommission, estimatedSlippage: row.estimatedSlippage, estimatedSpread: row.estimatedSpread }
    : ZERO_TRIPLE;
  const evaluation = evaluateSetupOutcome(
    {
      created_at: new Date(row.time).toISOString(),
      limit_entry: setup.entry,
      side: setup.side,
      stop_loss: setup.stop,
      symbol,
      take_profit: setup.target,
      take_profit_1: setup.tp1,
    },
    bars,
    now,
    {
      ...resolverCostOptions(triple, charged ? scale : 0),
      barIntervalMs: intervalMs,
      reviewHours: cal.reviewHours,
      runnerProtection: cal.runnerProtection,
      sameBarProtectionArming: arming,
      streamStartsAtMs: row.time + FIFTEEN,
    },
  );
  if (evaluation.state !== "resolved") {
    throw new Error(`mirror: ${symbol} ${row.time} ${setup.side} did not resolve (state ${evaluation.state})`);
  }
  const R = realizedRFromLegs({
    legs: evaluation.legs,
    perLegCost: charged ? triple.estimatedCommission / 2 : 0,
    riskDistance: Math.abs(setup.entry - setup.stop),
    side: setup.side,
  });
  return {
    R,
    filled: evaluation.outcome !== "unfilled",
    outcome: evaluation.outcome,
    legs: evaluation.legs,
    feedback: evaluation.feedback,
    exitAt: evaluation.exitAt,
    filledAt: evaluation.filledAt,
  };
}

// Aggregation — only reached by an anchored reader.

type Bound = { mean: number; lo: number | null; hi: number | null; clusters: number; n: number };
function plainMean(values: Array<{ cluster: number; value: number }>): Bound {
  const b = clusteredBound(values);
  return { mean: b.mean, lo: b.lower, hi: b.upper, clusters: b.clusters, n: b.n };
}
/**
 * Per-fill figures as ratios of sums, CR1 day-clustered by linearisation:
 * r1 = ΣS/nS, r2 = ΣM/nM; `kind` "S" | "M" | "diff" (r1 - r2) | "half" ((r1 + r2)/2).
 * Influence per day: u = a·(S_d - r1·nS_d)/nS + b·(M_d - r2·nM_d)/nM, SE² = G/(G-1)·Σu².
 */
function perFill(rows: Result[], arm: ArmKey, kind: "S" | "M" | "diff" | "half"): Bound {
  const days = new Map<number, [number, number, number, number]>();
  let sS = 0, nS = 0, sM = 0, nM = 0;
  for (const r of rows) {
    const d = days.get(r.day) ?? [0, 0, 0, 0];
    if (r.Sf[arm]) { d[0] += r.S[arm]; d[1] += 1; sS += r.S[arm]; nS += 1; }
    if (r.Mf[arm]) { d[2] += r.M[arm]; d[3] += 1; sM += r.M[arm]; nM += 1; }
    days.set(r.day, d);
  }
  const [a, b] = kind === "S" ? [1, 0] : kind === "M" ? [0, 1] : kind === "diff" ? [1, -1] : [0.5, 0.5];
  if ((a !== 0 && nS === 0) || (b !== 0 && nM === 0)) return { mean: Number.NaN, lo: null, hi: null, clusters: 0, n: 0 };
  const r1 = nS > 0 ? sS / nS : 0;
  const r2 = nM > 0 ? sM / nM : 0;
  const mean = a * r1 + b * r2;
  // Clusters: days holding a fill on a side the statistic reads.
  let G = 0;
  let squares = 0;
  for (const d of days.values()) {
    if ((a !== 0 && d[1] > 0) || (b !== 0 && d[3] > 0)) G += 1;
    const u = (a !== 0 ? (a * (d[0] - r1 * d[1])) / nS : 0) + (b !== 0 ? (b * (d[2] - r2 * d[3])) / nM : 0);
    squares += u * u;
  }
  if (G < 2) return { mean, lo: null, hi: null, clusters: G, n: a !== 0 && b !== 0 ? nS + nM : a !== 0 ? nS : nM };
  const se = Math.sqrt((G / (G - 1)) * squares);
  const margin = tMultiplier95(G - 1) * se;
  return { mean, lo: mean - margin, hi: mean + margin, clusters: G, n: a !== 0 && b !== 0 ? nS + nM : a !== 0 ? nS : nM };
}
function perDecision(rows: Result[], arm: ArmKey, kind: "S" | "M" | "diff" | "half"): Bound {
  return plainMean(rows.map((r) => ({
    cluster: r.day,
    value: kind === "S" ? r.S[arm] : kind === "M" ? r.M[arm] : kind === "diff" ? r.S[arm] - r.M[arm] : (r.S[arm] + r.M[arm]) / 2,
  })));
}
const f = (v: number | null, w = 8) => (v === null || !Number.isFinite(v) ? "-" : (v >= 0 ? "+" : "") + v.toFixed(4)).padStart(w);
const cell = (b: Bound) => `${f(b.mean)} [${f(b.lo, 0)}, ${f(b.hi, 0)}]`;

const decisionsAll = results.filter((r) => !r.noBars);
const noBarsCount = results.length - decisionsAll.length;
type CellKey = { split: string; years: string; pool: string; span: string };
function select(key: CellKey, pool: Result[] = decisionsAll): Result[] {
  return pool.filter((r) =>
    r.split === key.split && r.years === key.years && (key.pool === "all" || r.pool === key.pool) &&
    (key.span === "all" || r.span === key.span)
  );
}
const STATS = ["S", "M", "diff", "half"] as const;
const LABEL: Record<(typeof STATS)[number], string> = { S: "S", M: "M", diff: "S-M", half: "(S+M)/2" };
const ARM_LABEL: Record<ArmKey, string> = {
  fricOn: "frictionless · arming ON",
  fricOff: "frictionless · arming OFF",
  fullOn: "full cost · arming ON",
  fullOff: "full cost · arming OFF",
};

const tsv: string[] = [
  ["split", "years", "pool", "span", "arm", "basis", "stat", "decisions", "S_fills", "M_fills", "n", "day_clusters", "mean", "lo95", "hi95"].join("\t"),
];
const text: string[] = [];
text.push(
  `MIRROR CONTROL — forex shipped ladder vs its same-geometry mirror (MEASURED: ${outDir}/table.tsv, from resolve.mts)`,
  `corpus ${corpus} · manifest ${String(manifest.manifestHash).slice(0, 12)} · anchor ${manifest.anchor} · analyzer ${manifest.analyzerVersion} (HEAD ${ANALYZER_VERSION})`,
  `population: forex (manifest assetType) · fit/select · accepted · variant baseline · data-absent rows (noBarsInReviewWindow) excluded: ${noBarsCount}`,
  `unfilled decisions count 0 R per decision; per fill = ratio of sums (S−M per fill = ΣS/nS − ΣM/nM; (S+M)/2 per fill = half their sum)`,
  `95% intervals: CR1 clustered by UTC decision day at tMultiplier95(G−1); per-fill ratios by linearisation`,
  "",
);
for (const split of ["fit", "select"]) {
  for (const years of ["contained", "escaping"]) {
    for (const pool of ["in-pool", "held-out", "all"]) {
      for (const span of ["all", "in", "out"]) {
        const rows = select({ split, years, pool, span });
        if (rows.length === 0) continue;
        const spanLabel = span === "all" ? "all hours" : span === "in" ? "16-21 UTC" : "outside 16-21";
        text.push(`== ${split} · ${years} · ${pool} · ${spanLabel} · decisions ${rows.length}`);
        for (const arm of ARMS) {
          const sF = rows.filter((r) => r.Sf[arm]).length;
          const mF = rows.filter((r) => r.Mf[arm]).length;
          const pd = STATS.map((s) => perDecision(rows, arm, s));
          const pf = STATS.map((s) => perFill(rows, arm, s));
          text.push(`  ${ARM_LABEL[arm].padEnd(26)} fills S ${sF} M ${mF}`);
          text.push(`    per decision  ` + STATS.map((s, i) => `${LABEL[s]} ${cell(pd[i])}`).join("   "));
          text.push(`    per fill      ` + STATS.map((s, i) => `${LABEL[s]} ${cell(pf[i])}`).join("   "));
          STATS.forEach((s, i) => {
            for (const [basis, b] of [["decision", pd[i]], ["fill", pf[i]]] as const) {
              tsv.push([split, years, pool, span, arm, basis, LABEL[s], rows.length, sF, mF, b.n, b.clusters, b.mean, b.lo ?? "", b.hi ?? ""].join("\t"));
            }
          });
        }
      }
    }
  }
}
// By S's side, for the contained in-pool all-hours cells: S−M that flips sign
// with side is drift meeting a side imbalance, not per-decision information.
text.push("", "== BY SHIPPED SIDE — contained · in-pool · all hours · frictionless, per decision (S−M and (S+M)/2)");
for (const split of ["fit", "select"]) {
  for (const side of ["buy", "sell"] as const) {
    const rows = select({ split, years: "contained", pool: "in-pool", span: "all" }).filter((r) => r.side === side);
    const parts = (["fricOff", "fricOn"] as ArmKey[]).map((arm) =>
      `${arm === "fricOn" ? "ON " : "OFF"}: S ${cell(perDecision(rows, arm, "S"))} M ${cell(perDecision(rows, arm, "M"))} S-M ${cell(perDecision(rows, arm, "diff"))}`
    );
    text.push(`  ${split} ${side.padEnd(4)} n ${String(rows.length).padStart(6)}  ${parts.join("  |  ")}`);
    for (const arm of ARMS) {
      for (const s of STATS) {
        const b = perDecision(rows, arm, s);
        tsv.push([split, "contained", "in-pool", `all/side=${side}`, arm, "decision", LABEL[s], rows.length, rows.filter((r) => r.Sf[arm]).length, rows.filter((r) => r.Mf[arm]).length, b.n, b.clusters, b.mean, b.lo ?? "", b.hi ?? ""].join("\t"));
      }
    }
  }
}
writeFileSync(`${outDir}/table.tsv`, tsv.join("\n") + "\n");
writeFileSync(`${outDir}/table.txt`, text.join("\n") + "\n");

// Per market: contained fit and contained select, frictionless S, M, S−M, per
// decision (and per fill beside it), arming OFF and ON.
const pm: string[] = [
  `PER MARKET — contained years, frictionless, per decision [CR1 day-clustered 95%] (MEASURED: ${outDir}/per-market.tsv)`,
  `pool: * = held-out`,
];
const pmTsv: string[] = [["split", "symbol", "pool", "arm", "basis", "stat", "decisions", "S_fills", "M_fills", "n", "day_clusters", "mean", "lo95", "hi95"].join("\t")];
for (const split of ["fit", "select"]) {
  pm.push("", `== ${split} · contained`);
  pm.push(`${"market".padEnd(8)} ${"n".padStart(6)}  ${"arming OFF: S".padEnd(30)} ${"M".padEnd(30)} ${"S-M".padEnd(30)} | ${"arming ON: S".padEnd(30)} ${"M".padEnd(30)} ${"S-M".padEnd(30)}`);
  const symbols = [...new Set(decisionsAll.map((r) => r.symbol))].sort();
  for (const symbol of symbols) {
    const rows = decisionsAll.filter((r) => r.symbol === symbol && r.split === split && r.years === "contained");
    if (rows.length === 0) continue;
    const pool = holdout.held.has(symbol) ? "held-out" : "in-pool";
    const cols: string[] = [];
    for (const arm of ["fricOff", "fricOn"] as ArmKey[]) {
      for (const s of ["S", "M", "diff"] as const) cols.push(cell(perDecision(rows, arm, s)).padEnd(30));
    }
    pm.push(`${(symbol + (pool === "held-out" ? "*" : "")).padEnd(8)} ${String(rows.length).padStart(6)}  ${cols.slice(0, 3).join(" ")} | ${cols.slice(3).join(" ")}`);
    for (const arm of ARMS) {
      for (const s of STATS) {
        for (const [basis, b] of [["decision", perDecision(rows, arm, s)], ["fill", perFill(rows, arm, s)]] as const) {
          pmTsv.push([split, symbol, pool, arm, basis, LABEL[s], rows.length, rows.filter((r) => r.Sf[arm]).length, rows.filter((r) => r.Mf[arm]).length, b.n, b.clusters, b.mean, b.lo ?? "", b.hi ?? ""].join("\t"));
        }
      }
    }
  }
}
writeFileSync(`${outDir}/per-market.tsv`, pmTsv.join("\n") + "\n");
writeFileSync(`${outDir}/per-market.txt`, pm.join("\n") + "\n");

];
for (const r of results) {

}
writeFileSync(`${outDir}/rows.tsv`, rowTsv.join("\n") + "\n");
console.log(`done in ${((Date.now() - started) / 1000).toFixed(0)}s; ${results.length} rows (${noBarsCount} data-absent excluded from the table)`);
