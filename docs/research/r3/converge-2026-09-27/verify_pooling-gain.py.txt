# Converge 2026-09-27, orchestrator's check: the pooling gain across the 15 crosses, from the refuter's
# per-day sums (refute-forex-crosses_reader.mts.txt writes days.json: per symbol|split|variant, rows of
# [day, n, net, bound, gross]). Gain = (sum of per-market SDs)^2 / variance of the pooled daily sum, over
# every day any cross traded (a market's untraded day counts 0). Usage: python3 this days.json
import json,sys,math,statistics as st
cells=json.load(open(sys.argv[1]))['cells']
X=["AUDCAD","AUDCHF","AUDNZD","CADCHF","EURAUD","EURCAD","EURCHF","EURGBP","EURNZD","GBPCAD","GBPCHF","GBPJPY","GBPNZD","NZDCAD","NZDCHF"]
for split in ('fit','select'):
  for arm,ix in (('net',2),('bound',3)):
    series={s:{r[0]:r[ix] for r in cells[f'{s}|{split}|base']} for s in X if f'{s}|{split}|base' in cells}
    days=sorted(set().union(*[set(v) for v in series.values()]))
    tot=[sum(series[s].get(t,0.0) for s in series) for t in days]
    sumsd=sum(math.sqrt(st.pvariance([series[s].get(t,0.0) for t in days])) for s in series)
    print(f"{split} all years {arm}: markets {len(series)} days {len(days)} pooling gain {sumsd**2/st.pvariance(tot):.2f}")
