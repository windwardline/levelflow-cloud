def tq(p, df):
    # Student t quantile by bisection on a numerically integrated CDF (no scipy).
    def cdf(x):
        # regularized incomplete beta via continued fraction
        if x == 0: return 0.5
        a, b = df / 2.0, 0.5
        xx = df / (df + x * x)
        def betacf(a, b, x):
            MAXIT, EPS, FPMIN = 300, 3e-14, 1e-300
            qab, qap, qam = a + b, a + 1, a - 1
            c, d = 1.0, 1 - qab * x / qap
            d = 1 / (d if abs(d) > FPMIN else FPMIN); h = d
            for m in range(1, MAXIT + 1):
                m2 = 2 * m
                aa = m * (b - m) * x / ((qam + m2) * (a + m2))
                d = 1 + aa * d; d = 1 / (d if abs(d) > FPMIN else FPMIN)
                c = 1 + aa / c; c = c if abs(c) > FPMIN else FPMIN
                h *= d * c
                aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2))
                d = 1 + aa * d; d = 1 / (d if abs(d) > FPMIN else FPMIN)
                c = 1 + aa / c; c = c if abs(c) > FPMIN else FPMIN
                de = d * c; h *= de
                if abs(de - 1) < EPS: break
            return h
        lb = math.lgamma(a + b) - math.lgamma(a) - math.lgamma(b) + a * math.log(xx) + b * math.log(1 - xx)
        ib = math.exp(lb) * betacf(a, b, xx) / a if xx < (a + 1) / (a + b + 2) else 1 - math.exp(lb) * betacf(b, a, 1 - xx) / b
        return 1 - 0.5 * ib if x >= 0 else 0.5 * ib
    lo, hi = -50.0, 50.0
    for _ in range(200):
        mid = (lo + hi) / 2
        if cdf(mid) < p: lo = mid
        else: hi = mid
    return (lo + hi) / 2

def stats(days, idx=1):
    vals = [v[idx] for v in days.values()]
    n = len(vals)
    if n < 2: return None
    mean = sum(vals) / n
    sd = statistics.stdev(vals)
    keys = sorted(int(k) for k in days)
    # lag-1 autocorrelation of consecutive-traded-day sums
    series = [days[str(k)][idx] for k in keys]
    if n > 2:
        m = mean
        num = sum((series[i] - m) * (series[i + 1] - m) for i in range(n - 1))
        den = sum((x - m) ** 2 for x in series)
        ac = num / den if den else 0.0
    else:
        ac = 0.0
    span = keys[-1] - keys[0] + 1
    fills = sum(v[0] for v in days.values())
    return dict(n=n, fills=fills, total=sum(vals), mean=mean, sd=sd, lo95=mean - tq(0.975, n - 1) * sd / math.sqrt(n), ac1=ac, span_days=span, density=n / span)

    return out
