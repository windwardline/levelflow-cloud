print('\n=== (b) month-matched contrast: escaping pair-months vs contained pair-months in the SAME calendar month (month-grain escape map)')
for sp in ('fit','select'):
    for popn,pop in (('28 forex',ALL),('15 crosses',X15)):
        by=defaultdict(lambda: {'esc':[], 'con':[]})
        for s in pop:
            for r in cl(s,sp):
                by[ymd(r[0])[1]][bucket_m(s,r[0])].append(r)
        for i,arm in ((2,'net'),(3,'bound')):
            num=den=var=0; me=[];mc=[]; months=0; ke=kc=0
            for M,g in by.items():
                if not g['esc'] or not g['con'] or len(g['esc'])<2 or len(g['con'])<2: continue
                e=[r[i] for r in g['esc']]; c=[r[i] for r in g['con']]
                w=len(e)*len(c)/(len(e)+len(c)); d=mean(e)-mean(c)
                num+=w*d; den+=w; var+=w*w*(stdev(e)**2/len(e)+stdev(c)**2/len(c)); months+=1; ke+=len(e); kc+=len(c)
                me.append((w,mean(e))); mc.append((w,mean(c)))
            if den==0: print(f'  {sp} {popn} {arm}: no mixed months'); continue
            est=num/den; se=math.sqrt(var)/den
            wm=lambda L: sum(w*x for w,x in L)/sum(w for w,_ in L)
            print(f'  {sp:6} {popn:10} {arm:5} mixed months {months:3d} esc clusters {ke:5d} con clusters {kc:6d} | weighted esc {wm(me):+.4f} con {wm(mc):+.4f} diff {est:+.4f} se~{se:.4f} z~{est/se:+.2f}')
