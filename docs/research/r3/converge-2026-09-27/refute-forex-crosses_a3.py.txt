print('\n=== cross-pair correlation of daily cluster R (same UTC day), 15 crosses')
for lab,sp,bf in (('select contained','select','con'),('select escaping','select','esc'),('fit contained','fit','con')):
    for i,arm in ((2,'net'),(3,'bound')):
        ser={s:{r[0]:r[i] for r in rows(s,sp) if bucket_y(s,r[0])==bf} for s in X15}
        cors=[]
        for a in range(len(X15)):
            for b in range(a+1,len(X15)):
                A,B=ser[X15[a]],ser[X15[b]]; days=sorted(set(A)&set(B))
                if len(days)<50: continue
                xa=[A[d] for d in days]; xb=[B[d] for d in days]
                ma,mb=mean(xa),mean(xb); sa,sb=stdev(xa),stdev(xb)
                cors.append(sum((x-ma)*(y-mb) for x,y in zip(xa,xb))/((len(days)-1)*sa*sb))
        rb=mean(cors); neff=15/(1+14*rb)
        # variance-ratio N_eff: var(sum over pairs per day) vs sum of vars
        allD=sorted(set().union(*[set(v) for v in ser.values()]))
        tot=[sum(ser[s].get(d,0.0) for s in X15) for d in allD]
        vs=sum(stdev([ser[s].get(d,0.0) for d in allD])**2 for s in X15)
        vr=stdev(tot)**2/vs
        print(f'  {lab:17} {arm:5} pairs {len(cors)} mean rho {rb:+.3f} (min {min(cors):+.3f} max {max(cors):+.3f}) N_eff(rho)={neff:.1f}  var(sum)/sum(var)={vr:.2f} -> N_eff(var)={15/vr:.1f}')
