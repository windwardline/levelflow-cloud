// run named test files, record pass/fail, restore, verify restored bytes.

const textMut = (file, a, b) => ({ files: [file], apply: () => {
  const p = `${ROOT}/${file}`; const s = readFileSync(p, "utf8");
  const n = s.split(a).length - 1; if (n !== 1) throw new Error(`anchor count ${n} in ${file}: ${a.slice(0, 60)}`);
  writeFileSync(p, s.replace(a, b));
} });
const jsonMut = (file, fn) => ({ files: [file], apply: () => {
  const p = `${ROOT}/${file}`; const j = JSON.parse(readFileSync(p, "utf8")); fn(j); writeFileSync(p, JSON.stringify(j, null, 2) + "\n");
} });
const copyMut = (file, from) => ({ files: [file], apply: () => copyFileSync(from, `${ROOT}/${file}`) });

const only = process.argv.slice(2);
for (const [name, mut, tests] of mutations) {
  if (only.length > 0 && !only.some((prefix) => name.startsWith(prefix))) continue;
  const before = mut.files.map((f) => [f, sha(`${ROOT}/${f}`), readFileSync(`${ROOT}/${f}`)]);
  let verdict;
  try {
    mut.apply();
    const changed = mut.files.some((f, i) => sha(`${ROOT}/${f}`) !== before[i][1]);
    if (!changed) { verdict = "NOT APPLIED (file unchanged)"; }
    else {
      const args = ["--no-install", "tsx", "--tsconfig", "tsconfig.tests.json", "--test", ...(tests.pattern ? [`--test-name-pattern=${tests.pattern}`] : []), ...tests.files];
      const r = spawnSync("npx", args, { cwd: ROOT, encoding: "utf8", timeout: 600_000 });
      const out = `${r.stdout}\n${r.stderr}`;
      const pass = /ℹ pass (\d+)/.exec(out)?.[1];
      const fail = /ℹ fail (\d+)/.exec(out)?.[1];
      const failing = [...out.matchAll(/^\s*✖ (.+?) \(\d/gm)].map((m) => m[1]).filter((t, i, all) => all.indexOf(t) === i && !/failing tests/.test(t));
      verdict = pass === undefined || fail === undefined ? `NO VERDICT (exit ${r.status})` : Number(fail) > 0 ? `KILLED (${fail} failing: ${failing.slice(0, 4).join(" | ")})` : `MISSED (${pass} pass)`;
    }
  } catch (error) {
    verdict = `HARNESS ERROR ${error.message.slice(0, 120)}`;
  } finally {
    for (const [f, h, bytes] of before) { writeFileSync(`${ROOT}/${f}`, bytes); if (sha(`${ROOT}/${f}`) !== h) throw new Error(`RESTORE FAILED ${f}`); }
  }
  console.log(`${name}: ${verdict}`);
}
